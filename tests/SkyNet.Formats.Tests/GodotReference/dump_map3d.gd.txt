## Regenerates godot_dump_map3d.txt: the Godot port's decode of every MAP and every .3D model,
## as counts, samples and MD5s. Run it against a scratch Godot project holding copies of the
## port's scripts/loaders/{bsa_reader,map_file,mesh_3d}.gd (with mesh_3d.gd's one
## `Render.style(...)` call replaced by `pass`, since a --script run has no autoloads):
##   SKYNET_DATA=<game data folder> DUMP_OUT=<file> godot --headless --path <scratch project> --script res://dump.gd
extends SceneTree

const BSAReader := preload("res://scripts/loaders/bsa_reader.gd")
const MapFile := preload("res://scripts/loaders/map_file.gd")
const Mesh3D := preload("res://scripts/loaders/mesh_3d.gd")

func _md5(b: PackedByteArray) -> String:
	var ctx := HashingContext.new()
	ctx.start(HashingContext.HASH_MD5)
	if not b.is_empty():
		ctx.update(b)
	return ctx.finish().hex_encode()

func _ent(e) -> String:
	return "%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d" % [
		e.file_off, e.cell_x, e.cell_z, e.x, e.y, e.z, e.flags, e.off_x, e.off_y, e.off_z,
		e.name_index, e.sprite_index, e.marker_type, e.enemy_type, int(e.convoy),
		e.light_intensity, e.light_enable, e.state_byte, e.link_next, e.link_act_type, e.hp,
		e.exit_map, e.exit_marker_id, e.link_off, e.destroy_type, e.destroy_param, int(e.uses_defaults)]

func _init() -> void:
	var data: String = OS.get_environment("SKYNET_DATA")
	var out: String = OS.get_environment("DUMP_OUT")
	var lines: PackedStringArray = []
	var bsa = BSAReader.new()
	bsa.open(data + "/MDMDMAP2.BSA", 2)
	for en in bsa.entries():
		var nm: String = en.name
		if not nm.begins_with("MAP.") or not nm.get_extension().is_valid_int():
			continue
		var bytes: PackedByteArray = bsa.read(nm)
		var m = MapFile.parse(bytes)
		if m == null:
			lines.append("MAPSUM %s null" % nm)
			continue
		var flag: int = bytes.decode_u32(9028)
		var all: PackedStringArray = []
		for e in m.entities:
			all.append(_ent(e))
		lines.append("MAPSUM %s %d %d %d %d %d %s %s" % [nm, m.grid_width, m.grid_height, m.names.size(),
			m.entities.size(), flag, _md5("\n".join(all).to_utf8_buffer()), _md5("|".join(m.names).to_utf8_buffer())])
		if nm in ["MAP.210", "MAP.231", "MAP.211"]:
			for i in m.names.size():
				lines.append("MAPNAME %s %d %s" % [nm, i, m.names[i] if m.names[i] != "" else "-"])
			for i in m.entities.size():
				lines.append("ENT %s %d %s" % [nm, i, _ent(m.entities[i])])
	bsa.close()
	for arc in ["MDMDOBJS.BSA", "MDMDENMS.BSA"]:
		var b = BSAReader.new()
		b.open(data + "/" + arc, 2)
		for en in b.entries():
			var nm: String = en.name
			if not nm.ends_with(".3D"):
				continue
			var mesh = Mesh3D.parse(b.read(nm), nm)
			if mesh == null:
				lines.append("MDL %s %s null" % [arc, nm])
				continue
			var vb := PackedByteArray()
			for fr in mesh.frames:
				vb.append_array(fr.to_byte_array())
			var fs: PackedStringArray = []
			for f in mesh.faces:
				fs.append("%d;%s;%s;%s" % [f.type, ",".join(Array(f.idx).map(func(x): return str(x))),
					",".join(Array(f.du).map(func(x): return str(x))), ",".join(Array(f.dv).map(func(x): return str(x)))])
			lines.append("MDL %s %s %d %d %d %s %s" % [arc, nm, mesh.vertices.size(), mesh.faces.size(), mesh.frames.size(),
				_md5(vb), _md5("|".join(fs).to_utf8_buffer())])
		b.close()
	var f := FileAccess.open(out, FileAccess.WRITE)
	f.store_string("\n".join(lines) + "\n")
	f.close()
	quit()
